const ISAFLIX_WEB_CACHE_KEY = 'isaflix-web-discovery-v1';

function isaFlixCatalog() {
    return window.__ISAFLIX_WEB_CATALOG || {movies:[],series:[]};
}

function isaFlixNormalizeTitle(value) {
    return String(value || '')
        .replace(/<[^>]*>/g,' ')
        .replace(/&nbsp;|&#160;/gi,' ')
        .replace(/&amp;/gi,'&')
        .replace(/&#39;|&apos;/gi,"'")
        .replace(/&quot;/gi,'"')
        .replace(/\s+/g,' ')
        .trim();
}

function isaFlixDiscoverType(text, sourceKind) {
    const value = String(text || '').toLocaleLowerCase('tr');
    if (/dizi|series|sezon|season|episode|bölüm|bolum|tv-show/.test(value))
        return 'series';
    if (/film|movie|sinema|full-hd/.test(value))
        return 'movie';
    return sourceKind === 'series' ? 'series' :
           sourceKind === 'movie' ? 'movie' : '';
}

function isaFlixPosterFromAnchor(anchor, baseUrl) {
    const image = anchor.querySelector('img') ||
        (anchor.parentElement && anchor.parentElement.querySelector('img'));
    if (!image) return '';
    const raw = image.getAttribute('data-src') ||
        image.getAttribute('data-lazy-src') ||
        image.getAttribute('data-original') ||
        image.getAttribute('src') || '';
    try {
        const url = new URL(raw, baseUrl);
        return /^https?:$/.test(url.protocol) ? url.href : '';
    } catch (e) {
        return '';
    }
}

async function isaFlixDiscoverOneSource(source) {
    const html = await nativeFetch(source.url);
    if (!html || html.length < 100) return [];

    const doc = new DOMParser().parseFromString(html, 'text/html');
    const base = new URL(source.url);
    const found = [];
    const seen = new Set();

    for (const anchor of Array.from(doc.querySelectorAll('a[href]')).slice(0, 1200)) {
        try {
            const page = new URL(anchor.getAttribute('href'), base.href);
            if (!/^https?:$/.test(page.protocol) || page.origin !== base.origin)
                continue;

            const path = page.pathname.toLocaleLowerCase('tr');
            if (path === '/' || path.length < 3 ||
                /login|register|uye-ol|giris|search|arama|category|kategori|tag|etiket|page\/\d+/i.test(path))
                continue;

            const image = isaFlixPosterFromAnchor(anchor, base.href);
            const title = isaFlixNormalizeTitle(
                anchor.getAttribute('title') ||
                anchor.getAttribute('aria-label') ||
                (anchor.querySelector('img') &&
                 anchor.querySelector('img').getAttribute('alt')) ||
                anchor.textContent
            );

            if (!title || title.length < 2 || title.length > 140)
                continue;

            const type = source.kind === 'movie'
                ? 'movie'
                : source.kind === 'series'
                    ? 'series'
                    : isaFlixDiscoverType(
                        path + ' ' + title + ' ' + (anchor.className || ''),
                        source.kind
                    );
            if (!type) continue;

            if (/^(izle|watch|detay|details|fragman|trailer|ana sayfa)$/i.test(title))
                continue;

            const key = type + '|' + page.href;
            if (seen.has(key)) continue;
            seen.add(key);

            found.push({
                id:'web-' + source.id + '-' + found.length,
                name:title,
                title:title,
                type:type,
                category_name:type === 'series' ? 'Diziler' : 'Filmler',
                platform_name:type === 'series' ? 'Diziler' : 'Filmler',
                poster_url:image,
                image_url:image,
                page_url:page.href,
                source_id:source.id,
                source_url:base.origin,
                episodes:[]
            });
        } catch (e) {}
    }

    return found;
}

async function isaFlixDiscoverWebCatalog() {
    const empty = {movies:[],series:[]};

    try {
        const cached = localStorage.getItem(ISAFLIX_WEB_CACHE_KEY);
        if (cached) {
            const parsed = JSON.parse(cached);
            if (parsed && Array.isArray(parsed.movies) &&
                Array.isArray(parsed.series)) {
                window.__ISAFLIX_WEB_CATALOG = parsed;
                window.dispatchEvent(new CustomEvent(
                    'isa:catalog-catalog-updated', {detail:parsed}
                ));
            }
        }
    } catch (e) {}

    const collected = [];
    let cursor = 0;
    const workers = Array.from({length:2}, async () => {
        while (cursor < ISAFLIX_WEB_SOURCES.length) {
            const source = ISAFLIX_WEB_SOURCES[cursor++];
            try {
                const items = await isaFlixDiscoverOneSource(source);
                collected.push(...items);
            } catch (e) {
                console.warn('İSAFLIX kaynak taraması tamamlanamadı:', source.id);
            }
        }
    });

    await Promise.all(workers);

    const unique = new Map();
    for (const item of collected) {
        const key = item.type + '|' +
            item.title.toLocaleLowerCase('tr')
                .replace(/[^a-z0-9çğıöşü]/gi,'');
        if (!unique.has(key)) unique.set(key, item);
        else {
            const previous = unique.get(key);
            if (!previous.poster_url && item.poster_url)
                previous.poster_url = item.poster_url;
        }
