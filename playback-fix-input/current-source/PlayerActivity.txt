    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        broadcastId = getIntent().getStringExtra("broadcast_id");
        if (broadcastId != null && !broadcastId.isEmpty()) {
            activeBroadcastPlayer = this;
        }


        getWindow().addFlags(
                WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON
        );

        String queueJson = getIntent().getStringExtra("queue_json");

String queueFile = getIntent().getStringExtra("queue_file");

if (queueFile != null) {
    try (java.io.FileInputStream input =
                 new java.io.FileInputStream(queueFile)) {

        java.io.ByteArrayOutputStream bytes =
                new java.io.ByteArrayOutputStream();

        byte[] buffer = new byte[8192];
        int count;

        while ((count = input.read(buffer)) != -1) {
            bytes.write(buffer, 0, count);
        }

        queueJson = bytes.toString("UTF-8");

    } catch (Exception e) {
        android.util.Log.e("ISATV", "Kanal listesi okunamadı", e);
        queueJson = "[]";
    }
}

        selectedIndex =
                getIntent().getIntExtra("selected_index", 0);

        // Ekran döndürülünce son izlenen kanalı geri yükle.
        if (savedInstanceState != null) {
            selectedIndex = savedInstanceState.getInt(
                    "isa_saved_channel_index", selectedIndex);
        }


        try {
            JSONArray array = new JSONArray(
                    queueJson == null ? "[]" : queueJson
            );

            for (int i = 0; i < array.length(); i++) {
                JSONObject object = array.getJSONObject(i);

                Channel channel = new Channel();
                channel.name = object.optString("name", "İsimsiz kanal");
                channel.url = object.optString("url", "");
                channel.group = object.optString("group", "Genel");
                channel.logo = object.optString("logo", "");
                channel.season = object.optInt("season", 0);
                channel.episode = object.optInt("episode", 0);
                channel.isLive = object.optBoolean("is_live", true);

                JSONObject h = object.optJSONObject("headers");

                if (h != null) {
                    Iterator<String> keys = h.keys();

                    while (keys.hasNext()) {
                        String key = keys.next();
                        String value = h.optString(key, "");

                        if (!value.isEmpty() &&
                                key.matches("(?i)[a-z0-9-]+") &&
                                !key.matches("(?i)host|content-length|connection|transfer-encoding|accept-encoding") &&
                                !value.contains("\r") && !value.contains("\n")) {
                            channel.headers.put(key, value);
                        }
                    }
                }

                if (channel.url.matches("(?i)^https?://.*")) {
                    channels.add(channel);
                }
            }

        } catch (Exception e) {
            Log.e(TAG, "Kanal listesi okunamadı", e);
        }

        if (channels.isEmpty()) {
            Channel single = new Channel();
            single.name = getIntent().getStringExtra("channel_name");
            single.url = getIntent().getStringExtra("stream_url");

            if (single.name == null) single.name = "İSA TV HD";

            String json = getIntent().getStringExtra("headers_json");

            try {
                JSONObject h = new JSONObject(json == null ? "{}" : json);
                Iterator<String> keys = h.keys();

                while (keys.hasNext()) {
                    String key = keys.next();
                    single.headers.put(key, h.optString(key, ""));
                }
            } catch (Exception ignored) {
            }

            if (single.url != null) channels.add(single);
        }

        if (channels.isEmpty()) {
            finish();
            return;
        }

        if (selectedIndex < 0 || selectedIndex >= channels.size()) {
            selectedIndex = 0;
        }

        createInterface();
        initializePlayer();
        playSelected();
    }

    @Override
    public boolean dispatchKeyEvent(android.view.KeyEvent event) {

        int key = event.getKeyCode();

        boolean next =
                key == android.view.KeyEvent.KEYCODE_DPAD_UP ||
                key == android.view.KeyEvent.KEYCODE_CHANNEL_UP;

        boolean previous =
                key == android.view.KeyEvent.KEYCODE_DPAD_DOWN ||
                key == android.view.KeyEvent.KEYCODE_CHANNEL_DOWN;

        if (next || previous) {

            if (event.getAction() == android.view.KeyEvent.ACTION_DOWN) {

                if (event.getRepeatCount() == 0) {
                    finishChannelSwipe(next ? 1 : -1);
                }
            }

            return true;
        }

        return super.dispatchKeyEvent(event);
    }

            private float startX;

            @Override
            public boolean onTouch(View v, android.view.MotionEvent event) {
                boolean portrait =
                        getResources().getConfiguration().orientation

                switch (event.getAction()) {
                    case android.view.MotionEvent.ACTION_DOWN:
                        startX = event.getX();

                        horizontalDrag = false;
                        ignoreHorizontal = false;
                        return true;

                    case android.view.MotionEvent.ACTION_MOVE:
                        if (channelTransitioning) return true;

                        float dx = event.getX() - startX;
                        float dy = event.getY() - startY;

                        if (Math.abs(dx) > dp(10) && Math.abs(dx) > Math.abs(dy)) {
                            dragging = true;
                            horizontalDrag = true;
                            float width = Math.max(1, playerView.getWidth());
                            float offset = Math.max(-width, Math.min(width, dx));
                            playerView.setTranslationX(offset);
                            if (portraitPanel != null) portraitPanel.setTranslationX(offset);
                        } else if (Math.abs(dy) > dp(8) && Math.abs(dy) > Math.abs(dx)) {
                            dragging = true;
                            horizontalDrag = false;
                            float height = Math.max(playerView.getHeight(), getResources().getDisplayMetrics().heightPixels);
                            float offset = Math.max(-height, Math.min(height, dy));
                            playerView.setTranslationY(offset);
                            if (portraitPanel != null) portraitPanel.setTranslationY(offset);
                        }
                        if (dragging) {
                            handler.removeCallbacks(hideChannelButton);
                            channelButton.setVisibility(View.GONE);
                        }
                        return true;

                    case android.view.MotionEvent.ACTION_UP:

                        float moveY = event.getY() - startY;
                        float moveX = event.getX() - startX;

                        if (dragging) {
                            if (horizontalDrag) {
                                float threshold = Math.max(dp(70), playerView.getWidth() * 0.18f);
                                if (Math.abs(moveX) >= threshold) finishChannelSwipe(moveX < 0 ? 1 : -1, true);
                                else animateViewsBack(true);
                            } else {
                                float threshold = Math.max(dp(90), playerView.getHeight() * 0.22f);
                                if (Math.abs(moveY) >= threshold) finishChannelSwipe(moveY < 0 ? 1 : -1, false);
                                else animateViewsBack(false);
                            }
                        } else if (Math.abs(moveX) < dp(30) &&
                                   Math.abs(moveY) < dp(30)) {
                            // Dokunma ile kanal listesi açılmaz.
                            // Kanal değiştirme kaydırma hareketi korunur.
                            channelButton.setVisibility(View.GONE);
                        }

                        dragging = false;
                        horizontalDrag = false;
                        return true;

                    case android.view.MotionEvent.ACTION_CANCEL:
                        if (dragging) animateViewsBack(horizontalDrag);
                        dragging = false;
                        horizontalDrag = false;
                        return true;
                }
                return true;
            }

    private ExoPlayer createManagedPlayer(boolean preload) {
        DefaultLoadControl loadControl =
                new DefaultLoadControl.Builder()
                        .setBufferDurationsMs(
                                15000,
                                60000,
                                1000,
                                3000
                        )
                        .setPrioritizeTimeOverSizeThresholds(true)
                        .build();

        ExoPlayer p = new ExoPlayer.Builder(this)
                .setLoadControl(loadControl)
                .build();

        p.setVolume(preload ? 0f : 1f);

        p.setTrackSelectionParameters(
                p.getTrackSelectionParameters()
                        .buildUpon()
                        .setMaxVideoBitrate(
                                preload ? 1500000 : Integer.MAX_VALUE
                        )
                        .build()
        );

        Handler retryHandler = new Handler(Looper.getMainLooper());
        final int[] retryCount = {0};

        Runnable[] retryTask = new Runnable[1];

        retryTask[0] = () -> {
            if (p != player || p.isReleased()) return;

            if (retryCount[0] >= 5) {
                Log.e(TAG, "Canlı yayın yeniden bağlanma denemeleri tükendi.");
                return;
            }

            retryCount[0]++;
            Log.w(TAG, "Yayın yeniden hazırlanıyor. Deneme: "
                    + retryCount[0]);

            p.prepare();
            p.play();
        };

        p.addListener(new Player.Listener() {
            @Override
            public void onPlaybackStateChanged(int state) {
                if (p != player) return;

                if (state == Player.STATE_READY) {
                    retryCount[0] = 0;
                    retryHandler.removeCallbacks(retryTask[0]);
                    status.setVisibility(View.GONE);
                    loadingSpinner.setVisibility(View.GONE);

                } else if (state == Player.STATE_BUFFERING) {
                    loadingSpinner.setVisibility(View.VISIBLE);

                } else if (state == Player.STATE_IDLE) {
                    loadingSpinner.setVisibility(View.GONE);

                } else if (state == Player.STATE_ENDED) {
                    MediaItem current = p.getCurrentMediaItem();

                    if (current != null
                            && current.localConfiguration != null) {
                        String uri = current.localConfiguration.uri
                                .toString().toLowerCase();

                        if (uri.contains(".m3u8")) {
                            retryHandler.removeCallbacks(retryTask[0]);
                            retryHandler.postDelayed(retryTask[0], 1500);
                        }
                    }
                }
            }

            @Override
            public void onPlayerError(PlaybackException error) {
                Log.e(TAG, "Oynatıcı hatası: "
                        + error.getErrorCodeName(), error);

                if (p == player) {
                    status.setVisibility(View.GONE);
                    loadingSpinner.setVisibility(View.VISIBLE);

                    retryHandler.removeCallbacks(retryTask[0]);

                    long delay = Math.min(
                            1500L * (retryCount[0] + 1),
                            10000L
                    );

                    retryHandler.postDelayed(retryTask[0], delay);
                }
            }
        });

        return p;
    }

    private void prepareChannel(ExoPlayer target, int index,
                                boolean preload) {
        if (index < 0 || index >= channels.size()) return;

        Channel channel = channels.get(index);

        try {
            DefaultHttpDataSource.Factory factory =
                    new DefaultHttpDataSource.Factory()
                            .setUserAgent(
                                    channel.headers.containsKey("User-Agent")
                                    ? channel.headers.get("User-Agent")
                                    : "Mozilla/5.0 (Linux; Android 11) "
                                    + "AppleWebKit/537.36 Chrome/120 Safari/537.36"
                            )
                            .setConnectTimeoutMs(20000)
                            .setReadTimeoutMs(60000)
                            .setAllowCrossProtocolRedirects(true)
                            .setDefaultRequestProperties(channel.headers);

            String url = channel.url;
            String lowerUrl = url.toLowerCase();
            boolean isHls = lowerUrl.contains(".m3u8");

            MediaItem.Builder itemBuilder = new MediaItem.Builder()
                    .setUri(Uri.parse(url))
                    .setMediaId(channel.name == null
                            ? url : channel.name);

            if (isHls) {
                itemBuilder.setLiveConfiguration(
                        new MediaItem.LiveConfiguration.Builder()
                                .setTargetOffsetMs(15000)
                                .setMinPlaybackSpeed(0.97f)
                                .setMaxPlaybackSpeed(1.03f)
                                .build()
                );
            }

            MediaItem item = itemBuilder.build();

            target.stop();
            target.clearMediaItems();

            target.setVolume(preload ? 0f : 1f);

            target.setTrackSelectionParameters(
                    target.getTrackSelectionParameters()
                            .buildUpon()
                            .setMaxVideoBitrate(
                                    preload ? 1500000 : Integer.MAX_VALUE
                            )
                            .build()
            );

            androidx.media3.exoplayer.source.MediaSource mediaSource;

            if (isHls) {
                mediaSource = new HlsMediaSource.Factory(factory)
                        .setAllowChunklessPreparation(true)
                        .createMediaSource(item);
            } else {
                mediaSource =
                        new androidx.media3.exoplayer.source.DefaultMediaSourceFactory(
                                factory
                        ).createMediaSource(item);
            }

            target.setMediaSource(mediaSource);
            target.prepare();
            target.setPlayWhenReady(!preload);

        } catch (Exception e) {
            Log.e(TAG, "Kanal hazırlanamadı: " + index, e);
        }
    }

    

    private ExoPlayer obtainStandbyPlayer() {
        if (!standbyPlayers.isEmpty()) {
            return standbyPlayers.remove(standbyPlayers.size() - 1);
        }

        return createManagedPlayer(true);
    }

    private void recyclePlayer(ExoPlayer p) {
        if (p == null || p == player) return;

        p.setPlayWhenReady(false);
        p.stop();
        p.clearMediaItems();
        p.setVolume(0f);

        if (!standbyPlayers.contains(p) && standbyPlayers.size() < 2) {
            standbyPlayers.add(p);
        } else if (!standbyPlayers.contains(p)) {
            p.release();
        }
    }

    private void prepareNeighborChannels() {
        if (channels.size() < 2 || player == null) return;

        int previous = (selectedIndex - 1 + channels.size())
                % channels.size();

        int next = (selectedIndex + 1) % channels.size();

        java.util.HashSet<Integer> wanted = new java.util.HashSet<>();
        wanted.add(previous);
        wanted.add(next);
        wanted.remove(selectedIndex);

        java.util.Iterator<Map.Entry<Integer, ExoPlayer>> iterator =
                preloadedPlayers.entrySet().iterator();

        while (iterator.hasNext()) {
            Map.Entry<Integer, ExoPlayer> entry = iterator.next();

            if (!wanted.contains(entry.getKey())) {
                ExoPlayer old = entry.getValue();
                iterator.remove();
                recyclePlayer(old);
            }
        }

        for (Integer index : wanted) {
            if (!preloadedPlayers.containsKey(index)) {
                ExoPlayer standby = obtainStandbyPlayer();

                preloadedPlayers.put(index, standby);
                prepareChannel(standby, index, true);
            }
        }
    }


    private void animateViewsBack(boolean horizontal) {
        if (playerView == null) return;

        playerView.animate().cancel();
        playerView.animate()
                .translationX(horizontal ? 0f : playerView.getTranslationX())
                .translationY(horizontal ? playerView.getTranslationY() : 0f)
                .setDuration(180)
                .start();

        if (portraitPanel != null) {
            portraitPanel.animate().cancel();
            portraitPanel.animate()
                    .translationX(horizontal ? 0f : portraitPanel.getTranslationX())
                    .translationY(horizontal ? portraitPanel.getTranslationY() : 0f)
                    .setDuration(180)
                    .start();
        }
    }

    private void finishChannelSwipe(int direction) {
        finishChannelSwipe(direction, false);
    }

    private void finishChannelSwipe(int direction, boolean horizontal) {
        if (channels.isEmpty() || channelTransitioning ||
                playerView == null) return;

        channelTransitioning = true;

        final int distance = horizontal ? Math.max(1, playerView.getWidth()) : Math.max(
                playerView.getHeight(), getResources().getDisplayMetrics().heightPixels);
        final float exit = direction > 0 ? -distance : distance;
        final float enter = -exit;

        playerView.animate().cancel();

        if (portraitPanel != null) {
            portraitPanel.animate().cancel();
        }

        playerView.animate()
                .translationX(horizontal ? exit : 0f)
                .translationY(horizontal ? 0f : exit)
                .setDuration(150)
                .withEndAction(() -> {
                    selectedIndex += direction;

                    if (selectedIndex >= channels.size()) selectedIndex = 0;
                    if (selectedIndex < 0) selectedIndex = channels.size() - 1;

                    playSelected();

                    playerView.setTranslationX(horizontal ? enter : 0f);
                    playerView.setTranslationY(horizontal ? 0f : enter);

                    if (portraitPanel != null) {
                        portraitPanel.setTranslationX(horizontal ? enter : 0f);
                        portraitPanel.setTranslationY(horizontal ? 0f : enter);
                    }

                    playerView.animate()
                            .translationX(0f)
                            .translationY(0f)
                            .setDuration(220)
                            .withEndAction(() -> {
                                channelTransitioning = false;
                            })
                            .start();

                    if (portraitPanel != null) {
                        portraitPanel.animate()
                                .translationX(0f)
                                .translationY(0f)
                                .setDuration(220)
                                .start();
                    }
                })
                .start();

        if (portraitPanel != null) {
            portraitPanel.animate()
                    .translationX(horizontal ? exit : 0f)
                    .translationY(horizontal ? 0f : exit)
                    .setDuration(150)
                    .start();
        }
    }

    
    private void playSelected() {
        if (player == null || selectedIndex < 0 ||
                selectedIndex >= channels.size()) return;

        Channel channel = channels.get(selectedIndex);

        updatePortraitOverlay();

        if (loadingSpinner != null) {
            loadingSpinner.setVisibility(View.VISIBLE);
        }

        if (status != null) {
            status.setVisibility(View.GONE);
        }

        ExoPlayer oldActive = player;
        ExoPlayer readyPlayer = preloadedPlayers.remove(selectedIndex);

        try {
            if (readyPlayer != null) {
                // Hazırlanmış komşu oynatıcıyı aktif hale getir.
                player = readyPlayer;
                playerView.setPlayer(player);

                player.setVolume(1f);
                player.setTrackSelectionParameters(
                        player.getTrackSelectionParameters()
                                .buildUpon()
                                .setMaxVideoBitrate(Integer.MAX_VALUE)
                                .build()
                );

                player.setPlayWhenReady(true);

                recyclePlayer(oldActive);

            } else {
                // Hazır değilse mevcut aktif oynatıcıyla aç.
                prepareChannel(player, selectedIndex, false);
            }

            setTitle(channel.name);
            status.setVisibility(View.GONE);

            // Önceden hazırlanmış oynatıcı READY olmuş olabilir.
            int activeState = player.getPlaybackState();

            if (activeState == Player.STATE_READY) {
                loadingSpinner.setVisibility(View.GONE);
            } else if (activeState == Player.STATE_BUFFERING ||
                       activeState == Player.STATE_IDLE) {
                loadingSpinner.setVisibility(View.VISIBLE);
            }

            applyPreloadPolicy();

        } catch (Exception e) {
            Log.e(TAG, "Kanal başlatılamadı", e);
            loadingSpinner.setVisibility(View.GONE);
        }
    }



    


private boolean isPortraitMode() {
    return getResources().getConfiguration().orientation
            == Configuration.ORIENTATION_PORTRAIT;
}

private boolean isTelevisionDevice() {
    int type = getResources().getConfiguration().uiMode
            & Configuration.UI_MODE_TYPE_MASK;

    return type == Configuration.UI_MODE_TYPE_TELEVISION
            || getPackageManager().hasSystemFeature(
                    android.content.pm.PackageManager.FEATURE_LEANBACK
            );
}

private boolean isPhoneDevice() {
    if (isTelevisionDevice()) return false;

    int smallestWidth =
            getResources().getConfiguration().smallestScreenWidthDp;

    if (smallestWidth > 0) {
        return smallestWidth < 600;
    }

    float density = getResources().getDisplayMetrics().density;
    float widthDp =
            getResources().getDisplayMetrics().widthPixels / density;

    return widthDp < 600;
}

private void stopNeighborPreloads() {
    Iterator<Map.Entry<Integer, ExoPlayer>> iterator =
            preloadedPlayers.entrySet().iterator();

    while (iterator.hasNext()) {
        ExoPlayer standby = iterator.next().getValue();
        iterator.remove();
        recyclePlayer(standby);
    }
}


    // Komşu yayınları aç ve altı saniyelik sayaç başlat.
    private void startNeighborPreloadsWithTimeout() {
        handler.removeCallbacks(neighborPreloadTimeout);

        neighborPreloadUnlocked = true;

        if (portraitWarning != null) {
            portraitWarning.setVisibility(View.GONE);
        }

        if (player != null && channels.size() >= 2) {
            prepareNeighborChannels();
        } else {
            stopNeighborPreloads();
        }

        handler.postDelayed(neighborPreloadTimeout, 6000L);
    }

private void applyPreloadPolicy() {
        startNeighborPreloadsWithTimeout();
    }

private boolean isCurrentChannelFavorite() {
    if (selectedIndex < 0 || selectedIndex >= channels.size()) return false;

    String url = channels.get(selectedIndex).url;
    try {
        JSONArray favorites = new JSONArray(
                getSharedPreferences("isa_tv_data", MODE_PRIVATE)
                        .getString("favorites", "[]"));

        for (int i = 0; i < favorites.length(); i++) {
            JSONObject item = favorites.optJSONObject(i);
            if (item != null && url.equals(item.optString("url", ""))) {
                return true;
            }
        }
    } catch (Exception e) {
        Log.e(TAG, "Favoriler okunamadı", e);
    }
    return false;
}

private void updatePortraitOverlay() {
    if (portraitChannelName == null || favoriteButton == null) return;

    boolean portrait = isPortraitMode();
    if (portraitPanel != null) portraitPanel.setVisibility(portrait ? View.VISIBLE : View.GONE);

    portraitChannelName.setVisibility(portrait ? View.VISIBLE : View.GONE);
    boolean liveItem = selectedIndex >= 0 && selectedIndex < channels.size()
            && channels.get(selectedIndex).isLive;
    portraitLiveBadge.setVisibility(portrait && liveItem ? View.VISIBLE : View.GONE);
    if (portraitEpisodeInfo != null) portraitEpisodeInfo.setVisibility(portrait ? View.VISIBLE : View.GONE);
    favoriteButton.setVisibility(portrait ? View.VISIBLE : View.GONE);

    if (!portrait) return;

    if (selectedIndex >= 0 && selectedIndex < channels.size()) {
        Channel channel = channels.get(selectedIndex);
        portraitChannelName.setText(
                channel.name == null ? "İSA TV HD" : channel.name);
        if (portraitEpisodeInfo != null) {
            if (channel.season > 0 || channel.episode > 0) {
                String label = (channel.season > 0 ? "Sezon " + channel.season : "") +
                        (channel.episode > 0 ? "  ·  " + channel.episode + ". Bölüm" : "");
                portraitEpisodeInfo.setText(label.trim());
                portraitEpisodeInfo.setVisibility(View.VISIBLE);
            } else portraitEpisodeInfo.setVisibility(View.GONE);
        }
    }

    favoriteButton.setText(isCurrentChannelFavorite() ? "♥" : "♡");
    favoriteButton.setContentDescription(
            isCurrentChannelFavorite()
                    ? "Favorilerden çıkar"
                    : "Favorilere ekle");
}

private void toggleCurrentFavorite() {
    if (selectedIndex < 0 || selectedIndex >= channels.size()) return;

    Channel channel = channels.get(selectedIndex);
    boolean add = !isCurrentChannelFavorite();

    try {
        JSONObject item = new JSONObject();
        item.put("name", channel.name == null ? "İsimsiz kanal" : channel.name);
        item.put("url", channel.url);
        item.put("group", channel.group == null ? "Genel" : channel.group);
        item.put("logo", channel.logo == null ? "" : channel.logo);
        item.put("is_live", channel.isLive);
        item.put("season", channel.season);
        item.put("episode", channel.episode);

        android.content.SharedPreferences prefs =
                getSharedPreferences("isa_tv_data", MODE_PRIVATE);

        JSONArray old = new JSONArray(prefs.getString("favorites", "[]"));
        JSONArray updated = new JSONArray();

        for (int i = 0; i < old.length(); i++) {
            JSONObject existing = old.optJSONObject(i);
            if (existing == null) continue;

            if (!channel.url.equals(existing.optString("url", ""))) {
                updated.put(existing);
            }
        }

        if (add) updated.put(item);

        prefs.edit().putString("favorites", updated.toString()).apply();
        updatePortraitOverlay();

    } catch (Exception e) {
        Log.e(TAG, "Favori değiştirilemedi", e);
    }
}

    @Override
    protected void onDestroy() {
        if (activeBroadcastPlayer == this) {
            activeBroadcastPlayer = null;
        }
        handler.removeCallbacksAndMessages(null);

        if (playerView != null) {
            playerView.setPlayer(null);
        }

        java.util.HashSet<ExoPlayer> allPlayers =
                new java.util.HashSet<>();

        if (player != null) allPlayers.add(player);
        allPlayers.addAll(preloadedPlayers.values());
        allPlayers.addAll(standbyPlayers);

        preloadedPlayers.clear();
        standbyPlayers.clear();

        for (ExoPlayer p : allPlayers) {
            try {
                p.release();
            } catch (Exception ignored) {}
        }

        player = null;

        super.onDestroy();
    }
