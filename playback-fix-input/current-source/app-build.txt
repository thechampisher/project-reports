// Debug APK başarılı derlemeden sonra İndirilenler klasörüne kopyalanır.
tasks.register("copyDebugApkToDownload") {
    doLast {
        def apk = layout.buildDirectory.file("outputs/apk/debug/app-debug.apk").get().asFile
        def destination = new File("/sdcard/Download/ISA-TV-HD.apk")
        if (!apk.isFile()) throw new GradleException("APK bulunamadı: " + apk)
        destination.parentFile.mkdirs()
        java.nio.file.Files.copy(apk.toPath(), destination.toPath(),
            java.nio.file.StandardCopyOption.REPLACE_EXISTING)
        println("APK kopyalandı: " + destination)
    }
}
tasks.matching { it.name == "assembleDebug" }.configureEach {
    finalizedBy("copyDebugApkToDownload")
}
